# @kaguerp/api-client@0.1.0

A TypeScript SDK client for the localhost API.

## Usage

First, install the SDK from npm.

```bash
npm install @kaguerp/api-client --save
```

Next, try it out.


```ts
import {
  Configuration,
  ExecutionScopeEndpointApi,
} from '@kaguerp/api-client';
import type { GetExecutionScopesRequest } from '@kaguerp/api-client';

async function example() {
  console.log("🚀 Testing @kaguerp/api-client SDK...");
  const config = new Configuration({
    // Configure HTTP bearer authorization: Bearer
    accessToken: "YOUR BEARER TOKEN",
  });
  const api = new ExecutionScopeEndpointApi(config);

  try {
    const data = await api.getExecutionScopes();
    console.log(data);
  } catch (error) {
    console.error(error);
  }
}

// Run the test
example().catch(console.error);
```
